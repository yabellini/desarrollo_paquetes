#' Dichos y expresiones populares de América Latina

#' Una colección de dichos, refranes y expresiones que se usan en distintos

#'   \item{significado}{Qué quiere decir, explicado en pocas palabras.}
#'   \item{pais}{País donde se usa. "Varios países" si se usa en muchos lugares.}
#'   \item{tema}{Tema del dicho: "dinero", "comida", "animales", etc.}

"dichos"

R/dicho_al_azar.R, versión con datos:

#' Elegir un dicho al azar
#'
#' Devuelve un dicho elegido al azar del conjunto de datos [dichos].
#' Podés elegir de qué país querés el dicho.
#'
#' @param pais Texto con el nombre de un país, por ejemplo `"Argentina"`.
#'   Si es `NULL` (el valor por defecto) se elige entre todos los países.
#'
#' @return Un texto (vector de caracteres de largo 1) con el dicho.
#' @export
#'
#' @examples
#' dicho_al_azar()
#' dicho_al_azar(pais = "Argentina")
dicho_al_azar <- function(pais = NULL) {
  opciones <- dichos

  if (!is.null(pais)) {
    opciones <- dichos[dichos$pais == pais, ]
  }

  fila <- sample(nrow(opciones), size = 1)
  opciones$dicho[fila]
}

Episodio 6: el error con cli

dicho_al_azar <- function(pais = NULL) {
  opciones <- dichos

  if (!is.null(pais)) {
    if (!pais %in% dichos$pais) {
      cli::cli_abort(c(
        "No tengo dichos de {.val {pais}}.",
        "i" = "Los pa\u00edses disponibles son: {.val {unique(dichos$pais)}}."
      ))
    }
    opciones <- dichos[dichos$pais == pais, ]
  }

  fila <- sample(nrow(opciones), size = 1)
  opciones$dicho[fila]
}

Extra: dicho_del_dia()

Una función que siempre da el mismo dicho para la misma fecha. Más sencillo de testear, porque no es al azar.

R/dicho_del_dia.R

#' El dicho del día
#'
#' Devuelve siempre el mismo dicho para una misma fecha. Ideal para
#' empezar la mañana (o una clase).
#'
#' @param fecha Una fecha (objeto de clase `Date`). Por defecto, hoy.
#'
#' @return Un texto con el dicho y su significado.
#' @export
#'
#' @examples
#' dicho_del_dia()
#' dicho_del_dia(as.Date("2026-12-03"))
dicho_del_dia <- function(fecha = Sys.Date()) {
  if (!inherits(fecha, "Date")) {
    cli::cli_abort(c(
      "{.arg fecha} tiene que ser una fecha.",
      "x" = "Recib\u00ed un objeto de clase {.cls {class(fecha)}}.",
      "i" = "Prob\u00e1 con {.code as.Date(\"2026-12-03\")}."
    ))
  }

  fila <- as.integer(fecha) %% nrow(dichos) + 1
  paste0(dichos$dicho[fila], ": ", dichos$significado[fila])
}

tests/testthat/test-dicho_del_dia.R

test_that("dicho_del_dia() da el mismo dicho para la misma fecha", {
  fecha <- as.Date("2026-12-03")
  expect_equal(dicho_del_dia(fecha), dicho_del_dia(fecha))
})

test_that("dicho_del_dia() cambia de un día al otro", {
  expect_false(
    dicho_del_dia(as.Date("2026-12-03")) == dicho_del_dia(as.Date("2026-12-04"))
  )
})

test_that("dicho_del_dia() pide una fecha", {
  expect_error(dicho_del_dia("hoy"), class = "rlang_error")
})

Extra: buscar_dicho()

R/buscar_dicho.R

#' Buscar dichos por palabra
#'
#' Busca una palabra en los dichos y en sus significados.
#'
#' @param palabra Texto a buscar. No distingue mayúsculas de minúsculas.
#'
#' @return Un data frame con las filas de [dichos] que contienen la palabra.
#'   Si no encuentra nada, devuelve un data frame sin filas y avisa con un
#'   mensaje.
#' @export
#'
#' @examples
#' buscar_dicho("dinero")
#' buscar_dicho("perro")
buscar_dicho <- function(palabra) {
  if (!is.character(palabra) || length(palabra) != 1) {
    cli::cli_abort("{.arg palabra} tiene que ser un solo texto.")
  }

  encontrados <- grepl(palabra, dichos$dicho, ignore.case = TRUE) |
    grepl(palabra, dichos$significado, ignore.case = TRUE)

  if (!any(encontrados)) {
    cli::cli_inform(c("!" = "No encontr\u00e9 dichos con {.val {palabra}}."))
  }

  dichos[encontrados, ]
}

Episodio 9: un pedido para un asistente de IA

Para probar con el chat de GitHub Copilot en https://github.com/copilot (gratis con tu cuenta de GitHub) o con otro asistente. Si usás Copilot en el navegador, elegí primero el repositorio de tu paquete para que pueda leerlo.

Agregá a dichosos la función `dichos_por_tema(tema)` en `R/dichos_por_tema.R`.
Usá R base, sin dependencias nuevas.
Si el tema no existe, dá un error con `cli::cli_abort()` que liste los temas disponibles, como en `dicho_al_azar()`.
Documentala con roxygen2 y exportala.
Escribí tests cortos en `tests/testthat/test-dichos_por_tema.R`.
Antes de escribir código, mostrame un plan.

Después: leé el plan, pedile el código, copialo a tu paquete, revisá los cambios con Git, probá la función con load_all() y corré devtools::check(). Y si compartís el resultado, contá que usaste IA.
